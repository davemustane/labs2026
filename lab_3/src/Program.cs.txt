using System;
using System.Collections.Generic;
using System.Linq;

namespace Notebook
{
    /// <summary>
    /// Точка входа в программу. Реализует консольное меню и взаимодействие с пользователем.
    /// </summary>
    class Program
    {
        private static List<Note> _notes = new List<Note>();
        private static int _nextId = 1;

        static void Main(string[] args)
        {
            _notes = NoteRepository.Load();
            _nextId = _notes.Count > 0 ? _notes.Max(n => n.Id) + 1 : 1;

            bool running = true;
            while (running)
            {
                ShowMenu();
                string? choice = Console.ReadLine()?.Trim();

                switch (choice)
                {
                    case "1": AddNote(); break;
                    case "2": ShowAllNotes(); break;
                    case "3": SearchNotes(); break;
                    case "4": DeleteNote(); break;
                    case "0":
                        NoteRepository.Save(_notes);
                        Console.WriteLine("Данные сохранены. До свидания!");
                        running = false;
                        break;
                    default:
                        Console.WriteLine("Неверный ввод. Выберите пункт от 0 до 4.");
                        break;
                }
            }
        }

        /// <summary>
        /// Выводит главное меню программы.
        /// </summary>
        private static void ShowMenu()
        {
            Console.WriteLine();
            Console.WriteLine("========== ЗАПИСНАЯ КНИЖКА ==========");
            Console.WriteLine("1. Добавить заметку");
            Console.WriteLine("2. Показать все заметки");
            Console.WriteLine("3. Поиск заметок");
            Console.WriteLine("4. Удалить заметку");
            Console.WriteLine("0. Выход (с сохранением)");
            Console.WriteLine("=====================================");
            Console.Write("Ваш выбор: ");
        }

        /// <summary>
        /// Добавляет новую заметку в коллекцию.
        /// </summary>
        private static void AddNote()
        {
            Console.Write("Заголовок заметки: ");
            string? title = Console.ReadLine()?.Trim();
            if (string.IsNullOrEmpty(title))
            {
                Console.WriteLine("[Ошибка] Заголовок не может быть пустым.");
                return;
            }

            Console.Write("Содержание заметки: ");
            string content = Console.ReadLine()?.Trim() ?? string.Empty;

            Console.Write("Теги (через запятую, можно оставить пустым): ");
            string? tagsInput = Console.ReadLine();
            var tags = ParseTags(tagsInput);

            var note = new Note
            {
                Id = _nextId++,
                Title = title,
                Content = content,
                CreatedAt = DateTime.Now,
                Tags = tags
            };

            _notes.Add(note);
            Console.WriteLine($"[OK] Заметка #{note.Id} успешно добавлена.");
        }

        /// <summary>
        /// Выводит все заметки из коллекции.
        /// </summary>
        private static void ShowAllNotes()
        {
            if (_notes.Count == 0)
            {
                Console.WriteLine("Записная книжка пуста.");
                return;
            }

            Console.WriteLine($"\n--- Всего заметок: {_notes.Count} ---");
            foreach (var note in _notes)
            {
                Console.WriteLine(note);
                Console.WriteLine();
            }
        }

        /// <summary>
        /// Ищет заметки по ключевому слову (в заголовке, содержимом и тегах).
        /// </summary>
        private static void SearchNotes()
        {
            Console.Write("Введите ключевое слово для поиска: ");
            string? keyword = Console.ReadLine()?.Trim();

            if (string.IsNullOrEmpty(keyword))
            {
                Console.WriteLine("[Ошибка] Ключевое слово не может быть пустым.");
                return;
            }

            var found = _notes
                .Where(n => n.Title.Contains(keyword, StringComparison.OrdinalIgnoreCase)
                         || n.Content.Contains(keyword, StringComparison.OrdinalIgnoreCase)
                         || n.Tags.Any(t => t.Contains(keyword, StringComparison.OrdinalIgnoreCase)))
                .ToList();

            if (found.Count == 0)
            {
                Console.WriteLine("Ничего не найдено.");
                return;
            }

            Console.WriteLine($"\n--- Найдено заметок: {found.Count} ---");
            foreach (var note in found)
            {
                Console.WriteLine(note);
                Console.WriteLine();
            }
        }

        /// <summary>
        /// Удаляет заметку по её идентификатору.
        /// </summary>
        private static void DeleteNote()
        {
            if (_notes.Count == 0)
            {
                Console.WriteLine("Записная книжка пуста — нечего удалять.");
                return;
            }

            Console.Write("Введите ID заметки для удаления: ");
            string? input = Console.ReadLine()?.Trim();

            if (!int.TryParse(input, out int id))
            {
                Console.WriteLine("[Ошибка] Необходимо ввести целое число.");
                return;
            }

            var note = _notes.FirstOrDefault(n => n.Id == id);
            if (note == null)
            {
                Console.WriteLine($"[Ошибка] Заметка с ID = {id} не найдена.");
                return;
            }

            _notes.Remove(note);
            Console.WriteLine($"[OK] Заметка #{id} удалена.");
        }

        /// <summary>
        /// Преобразует строку с тегами в список (разделитель — запятая).
        /// </summary>
        private static List<string> ParseTags(string? input)
        {
            if (string.IsNullOrWhiteSpace(input))
                return new List<string>();

            return input
                .Split(',', StringSplitOptions.RemoveEmptyEntries | StringSplitOptions.TrimEntries)
                .ToList();
        }
    }
}
